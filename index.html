import json import os def add_new_questions_safely(new_batch_questions): json_file_path = "questions.json" # 1. मौजूदा फाईल लोड करें if os.path.exists(json_file_path): with open(json_file_path, "r", encoding="utf-8") as f: data = json.load(f) else: data = [] # 2. सारे मौजूदा प्रश्नों का एक सेट बनाएं (डुप्लीकेट रोकने के लिए) existing_questions = {q["question"].strip().lower() for q in data if "question" in q} added_count = 0 skipped_count = 0 # 3. नया बैच चेक करके जोड़ें for q in new_batch_questions: q_text = q.get("question", "").strip().lower() if q_text and q_text not in existing_questions: data.append(q) existing_questions.add(q_text) # सेट में भी जोड़ें ताकि इसी बैच में आपस में रिपीट न हो added_count += 1 else: skipped_count += 1 # 4. फाईल सेव करें with open(json_file_path, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) print(f"सफलतापूर्वक {added_count} नए यूनिक प्रश्न जोड़े गए।") print(f"कुल {skipped_count} डुप्लीकेट प्रश्न ऑटोमैटिकली स्किप (रद्द) कर दिए गए।")
